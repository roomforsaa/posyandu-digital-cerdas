# Format TXT alternatif untuk food-keywords.
# Baris diawali "#" atau "##" = nama kategori (bebas: healthy, unhealthy,
# neutral, dangerous, atau kategori custom baru misal "allergen").
# Baris kosong / diawali "//" diabaikan (komentar).
#
# Format tiap baris kata kunci:
#   istilah
#   istilah | note: keterangan
#   istilah | note: keterangan | severity: high | minAgeMonths: 12
#
# File ini hanya CONTOH tambahan (delta) — akan digabung (merge) dengan
# food-keywords.json memakai mergeFoodKeywordConfig(), jadi tidak perlu
# menulis ulang semua kata kunci yang sudah ada.
